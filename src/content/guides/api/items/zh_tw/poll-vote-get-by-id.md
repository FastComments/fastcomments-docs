[api-resource-header-start name = 'PollVote'; route = 'GET /api/v1/poll-votes/:id'; creditsCost = 1; api-resource-header-end]

根據其 ID 讀取單一投票。

匿名投票的投票無法被讀取，請求會因 `poll-anonymous` 錯誤而失敗。請參閱 `PollVote` 結構以了解投票的 `privacy` 設定如何套用。

[inline-code-attrs-start title = 'PollVote 取得 cURL 範例'; type = 'bash'; useDemoTenant = true; isFunctional = false; inline-code-attrs-end]
[inline-code-start]
curl --request GET \
  --url 'https://fastcomments.com/api/v1/poll-votes/my-id?tenantId=demo&API_KEY=DEMO_API_SECRET'
[inline-code-end]

[inline-code-attrs-start title = 'PollVote 取得 請求結構'; type = 'typescript'; isFunctional = false; inline-code-attrs-end]
[inline-code-start]
interface PollVoteGetQueryParams {
    tenantId: string
    API_KEY: string
}
[inline-code-end]

[inline-code-attrs-start title = 'PollVote 取得 回應結構'; type = 'typescript'; isFunctional = false; inline-code-attrs-end]

interface PollVoteGetResponse {
    status: 'success' | 'failed'

    /** Included on failure. **/
    reason?: string
    pollVote: PollVote
}
